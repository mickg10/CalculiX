cmake_minimum_required(VERSION 3.22...4.2)
project(metal_example_tt_gmg_solve)
add_executable(metal_example_tt_gmg_solve)
target_sources(metal_example_tt_gmg_solve PRIVATE tt_gmg_solve.cpp gmg_solve.cpp)
if(NOT TARGET TT::Metalium)
    find_package(TT-Metalium REQUIRED)
endif()
find_package(OpenMP)
target_link_libraries(metal_example_tt_gmg_solve PUBLIC TT::Metalium lapack blas)
if(OpenMP_CXX_FOUND)
    target_link_libraries(metal_example_tt_gmg_solve PUBLIC OpenMP::OpenMP_CXX)
endif()
set_source_files_properties(gmg_solve.cpp PROPERTIES COMPILE_OPTIONS "-Wno-error;-fopenmp=libgomp;-I/usr/lib/gcc/x86_64-linux-gnu/11/include")
target_link_libraries(metal_example_tt_gmg_solve PUBLIC /usr/lib/x86_64-linux-gnu/libgomp.so.1)
