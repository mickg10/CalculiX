cmake_minimum_required(VERSION 3.22...4.2)
project(metal_example_reduce_fp32)
add_executable(metal_example_reduce_fp32)
target_sources(metal_example_reduce_fp32 PRIVATE reduce_fp32.cpp)
if(NOT TARGET TT::Metalium)
    find_package(TT-Metalium REQUIRED)
endif()
target_link_libraries(metal_example_reduce_fp32 PUBLIC TT::Metalium)
if(TARGET TT::CommonPCH)
    tt_reuse_precompile_headers(metal_example_reduce_fp32 TT::CommonPCH)
endif()
